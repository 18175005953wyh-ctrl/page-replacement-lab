cmake_minimum_required(VERSION 3.20)
project(PageReplacementLab VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(STRICT_WARNINGS "Treat warnings as errors" OFF)

add_library(page_core src/trace.c src/replacement.c src/report.c)
target_include_directories(page_core PUBLIC include)
add_executable(page_replacement_lab src/main.c)
target_link_libraries(page_replacement_lab PRIVATE page_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(page_tests tests/test_runner.c)
    target_link_libraries(page_tests PRIVATE page_core)
    add_test(NAME core_tests COMMAND page_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    set_tests_properties(core_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME cli_tests COMMAND ${CMAKE_COMMAND}
        "-DAPP=$<TARGET_FILE:page_replacement_lab>" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-fixtures" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.cmake")
endif()

foreach(target page_core page_replacement_lab page_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /utf-8)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
            if(STRICT_WARNINGS)
                target_compile_options(${target} PRIVATE /WX)
            endif()
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
            if(STRICT_WARNINGS)
                target_compile_options(${target} PRIVATE -Werror)
            endif()
        endif()
    endif()
endforeach()
